function doGet(e) { var sheetName = "總表"; var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName); var bookedDates = []; if (sheet) { var data = sheet.getDataRange().getValues(); // 我們印出第一列跟第二列來看看 F 欄到底是第幾欄、資料長怎樣 var debugInfo = { row2: data[1], // 第二列的資料 dateInF: data[1][5] // F 欄的值 (索引 5) }; var dateColIndex = 5; // F 欄 for (var i = 1; i < data.length; i++) { var val = data[i][dateColIndex]; if (val) { var d = new Date(val); if (!isNaN(d.getTime())) { var year = d.getFullYear(); var month = ('0' + (d.getMonth() + 1)).slice(-2); var day = ('0' + d.getDate()).slice(-2); bookedDates.push(year + '-' + month + '-' + day); } } } } // 如果抓不到任何日期，把 debug 資訊回傳出來讓我們看 var output = bookedDates.length > 0 ? bookedDates : ["DEBUG: 抓不到資料！", debugInfo]; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); }
